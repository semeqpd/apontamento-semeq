2. JavaScript - Event listener para mudança de status (change event)
   Arquivo: templates/apontamentos/lista.html
   LINHAS: 445-473

   Código:
   ----------------------------------------------------------------
   document.querySelectorAll('.select-status-rapido').forEach(select => {
       // Armazena o valor anterior para reverter se o usuário cancelar no modal
       select.addEventListener('focus', function() {
           statusAnteriorValue = this.value;
       });

       select.addEventListener('change', function(e) {
           e.stopPropagation();
           selectAtual = this;
           apontamentoIdAtual = this.dataset.apontamentoId;
           const novoStatusId = this.value;
           const optionSelecionada = this.options[this.selectedIndex];
           const novaCor = optionSelecionada.dataset.cor;
           const nomeStatus = optionSelecionada.text.trim().toLowerCase();

           // 1. Atualização em TEMPO REAL no frontend (Feedback instantâneo ao usuário)
           if (novaCor) {
               this.style.backgroundColor = novaCor;
           }

           // 2. Persistência no Banco via API
           fetch(`/apontamentos/${apontamentoIdAtual}/alterar-status/`, {
               method: 'POST',
               headers: {
                   'Content-Type': 'application/json',
                   'X-CSRFToken': '{{ csrf_token }}'
               },
               body: JSON.stringify({ status: novoStatusId })
           })
           .then(async response => {
               const data = await response.json();
               if (response.ok && data.success) {
                   // Status atualizado com sucesso - recarrega para atualizar totais
                   location.reload();
               } else if (data.requires_time) {
                   // Backend exigiu o tempo investido -> Abre o Modal
                   document.getElementById('inputTempoMinutos').value = data.tempo_atual || '';
                   if (modalTempo) modalTempo.show();
               } else {
                   alert('Erro: ' + (data.error || 'Falha ao atualizar status'));
                   if (statusAnteriorValue) selectAtual.value = statusAnteriorValue;
               }
           })
           .catch(err => {
               alert('Erro de conexão ao alterar status.');
               if (statusAnteriorValue) selectAtual.value = statusAnteriorValue;
           });
       });
   });

3. Event listener para salvar tempo no modal de conclusão
   Arquivo: templates/apontamentos/lista.html
   LINHAS: 476-531

   Código:
   ----------------------------------------------------------------
   const btnSalvarTempo = document.getElementById('btnSalvarTempoConclusao');
   if (btnSalvarTempo) {
       btnSalvarTempo.addEventListener('click', function() {
           const inputTempo = document.getElementById('inputTempoMinutos');
           const tempoStr = inputTempo.value.trim();
           
           // Validação: valor vazio
           if (tempoStr === '') {
               inputTempo.classList.add('is-invalid');
               return;
           }
           
           // Validação: deve ser número inteiro
           if (!/^\d+$/.test(tempoStr)) {
               inputTempo.classList.add('is-invalid');
               return;
           }
           
           const tempoValor = parseInt(tempoStr, 10);
           
           // Validação: zero ou negativo
           if (tempoValor <= 0) {
               inputTempo.classList.add('is-invalid');
               return;
           }
           
           inputTempo.classList.remove('is-invalid');
   
           if (modalTempo) modalTempo.hide();
   
           // Reenvia a requisição com o tempo preenchido no modal
           if (apontamentoIdAtual && selectAtual) {
               const novoStatusId = selectAtual.value;
               fetch(`/apontamentos/${apontamentoIdAtual}/alterar-status/`, {
                   method: 'POST',
                   headers: {
                       'Content-Type': 'application/json',
                       'X-CSRFToken': '{{ csrf_token }}'
                   },
                   body: JSON.stringify({
                       status: novoStatusId,
                       tempo_investido_minutos: tempoValor
                   })
               })
               .then(res => res.json())
               .then(data => {
                   if (data.success) {
                       location.reload();
                   } else {
                       alert(data.error || 'Erro ao atualizar.');
                       if (statusAnteriorValue && selectAtual) selectAtual.value = statusAnteriorValue;
                   }
               });
           }
       });
   });

4. Event listener para cancelar modal de tempo (reverte seleção)

   // Se cancelar o modal, reverte a seleção do dropdown para o status anterior
   ['btnFecharModalTempo', 'btnCancelarModalTempo'].forEach(id => {
       const btn = document.getElementById(id);
       if (btn) {
           btn.addEventListener('click', function() {
               if (selectAtual && statusAnteriorValue) {
                   selectAtual.value = statusAnteriorValue;
               }
           });
       });
   });
