3. JavaScript - Função setDeleteForm (preenche o formulário de exclusão)
   Arquivo: templates/apontamentos/lista.html
   LINHAS: 397-407

   Código:
   ----------------------------------------------------------------
   window.setDeleteForm = function(apontamentoPk) {
       document.getElementById('excluirApontamentoPk').value = apontamentoPk;
       const form = document.getElementById('formExcluirApontamento');
       // Use the form's current action and replace the PK placeholder
       const baseAction = form.getAttribute('action').replace('0/excluir/', '');
       form.setAttribute('action', baseAction + apontamentoPk + '/excluir/');
       document.getElementById('excluirApontamentoInfo').textContent = 'Apontamento #' + apontamentoPk;
   };
   ----------------------------------------------------------------
   LINHAS: 397-407

4. Event listener do modal (show.bs.modal) - preenche formulário ao abrir modal
   Arquivo: templates/apontamentos/lista.html
   LINHAS: 409-448

   Código:
   ----------------------------------------------------------------
   const modalExcluir = document.getElementById('modalExcluirApontamento');
   if (modalExcluir) {
       modalExcluir.addEventListener('show.bs.modal', function (event) {
           const button = event.relatedTarget;
           const apontamentoPk = button.getAttribute('data-pk');
           document.getElementById('excluirApontamentoPk').value = apontamentoPk;
           const form = document.getElementById('formExcluirApontamento');
           // Update form action with correct pk
           form.action = form.action.replace('/0/excluir/', '/' + apontamentoPk + '/excluir/');
           document.getElementById('excluirApontamentoInfo').textContent = 'Apontamento #' + apontamentoPk;
       });
   });
   ----------------------------------------------------------------
   LINHAS: 409-448
